+# ==========================================================================
+# Surcouche du lab pour le serveur d'intégration continue (VPS 2)
+#
+# Utilisée avec docker-compose.yml (COMPOSE_FILE dans .env) :
+#   - Jenkins et SonarQube ne publient plus aucun port : ils sont servis
+#     en HTTPS par Caddy (jenkins.tanguydavid.fr, sonar.tanguydavid.fr) ;
+#   - limites mémoire : en cas de dépassement, seul le conteneur fautif est
+#     arrêté, la mémoire du serveur reste disponible pour les autres services.
+# ==========================================================================
+
+services:
+  sonarqube:
+    ports: !reset []
+    mem_limit: 3g
+    environment:
+      SONAR_SEARCH_JAVAOPTS: "-Xms512m -Xmx512m"
+      SONAR_WEB_JAVAOPTS: "-Xmx512m"
+      SONAR_CE_JAVAOPTS: "-Xmx512m"
+
+  sonar-db:
+    mem_limit: 512m
+
+  jenkins:
+    ports: !reset []
+    mem_limit: 3g
+    environment:
+      JAVA_OPTS: -Xmx1g
+
+  # Reverse proxy HTTPS (certificats Let's Encrypt automatiques)
+  caddy:
+    image: caddy:2
+    restart: unless-stopped
+    ports:
+      - "80:80"
+      - "443:443"
+      - "443:443/udp"
+    volumes:
+      - ./Caddyfile:/etc/caddy/Caddyfile:ro
+      - caddy_data:/data
+      - caddy_config:/config
+    depends_on:
+      - jenkins
+      - sonarqube
+    mem_limit: 256m
+

+  caddy_data:
+  caddy_config:

+FROM jenkins/jenkins:lts-jdk21
+
+USER root
+
+# Outils dont la pipeline a besoin :
+# - Docker CLI + buildx : docker build / push / scan Trivy (via le daemon de l'hôte)
+# - Node.js 22 : npm ci, prisma generate, tests
+RUN apt-get update \
+ && apt-get install -y --no-install-recommends ca-certificates curl gnupg \
+ && install -m 0755 -d /etc/apt/keyrings \
+ && curl -fsSL https://download.docker.com/linux/debian/gpg -o /etc/apt/keyrings/docker.asc \
+ && echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] https://download.docker.com/linux/debian $(. /etc/os-release && echo $VERSION_CODENAME) stable" \
+      > /etc/apt/sources.list.d/docker.list \
+ && curl -fsSL https://deb.nodesource.com/setup_22.x | bash - \
+ && apt-get install -y --no-install-recommends docker-ce-cli docker-buildx-plugin nodejs \
+ && rm -rf /var/lib/apt/lists/*
+
+# Donne à l'utilisateur jenkins l'accès au socket Docker de l'hôte (même GID que le groupe docker de l'hôte)
+ARG DOCKER_GID=963
+RUN groupadd -g ${DOCKER_GID} docker-host && usermod -aG docker-host jenkins
+
+USER jenkins
+
+COPY plugins.txt /usr/share/jenkins/ref/plugins.txt
+RUN jenkins-plugin-cli --plugin-file /usr/share/jenkins/ref/plugins.txt
